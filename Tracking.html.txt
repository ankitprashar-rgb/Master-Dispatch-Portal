<script>
  const $ = id => document.getElementById(id);

  function hasGoogle(){
    return (typeof google !== 'undefined') &&
           google.script &&
           google.script.run;
  }

  // --------- LOAD DISPATCHES (DATE RANGE) ----------
  function loadRange(){
    const from = $('fromDate').value;
    let to     = $('toDate').value;

    const chip = $('summaryChip');
    chip.classList.add('hidden');
    chip.textContent = '';

    if (!from && !to){
      alert('Select at least a From Date.');
      return;
    }

    if (!to) to = from; // single-day search

    $('status').textContent = 'Loading…';
    $('status').className   = 'muted dots';

    google.script.run
      .withSuccessHandler(res=>{
        console.log('api_listDispatchesByRange_v1 raw response:', res);
        $('status').textContent = '';
        $('status').className   = 'muted';
        const tb = $('tbody');
        tb.innerHTML = '';

        if (!res || typeof res !== 'object') {
          $('listWrap').classList.add('hidden');
          alert('No response from server.');
          return;
        }

        if (res.ok === false) {
          $('listWrap').classList.add('hidden');
          alert(res.msg ? ('Error: ' + res.msg) : 'No dispatches found.');
          return;
        }

        const rowsRaw = Array.isArray(res.rows) ? res.rows : [];
        const rows = rowsRaw.filter(r => r && r.dispatchId && String(r.dispatchId).trim() !== '');

        if (!rows.length) {
          $('listWrap').classList.add('hidden');
          alert('No dispatches (with Dispatch ID) found in this range.');
          return;
        }

        // ✅ Show summary chip
        const count = rows.length;
        const label = count === 1 ? 'dispatch' : 'dispatches';
        chip.textContent = `Showing ${count} ${label} from ${from} to ${to}`;
        chip.classList.remove('hidden');

        $('listWrap').classList.remove('hidden');

        rows.forEach(row=>{
          const tr = document.createElement('tr');
          tr.innerHTML = `
            <td>${row.date        || ''}</td>
            <td>${row.clientName  || ''}</td>
            <td>${row.project     || ''}</td>
            <td>${row.dispatchId  || ''}</td>
            <td><input type="text" value="${row.trackingId||''}" placeholder="Tracking ID"></td>
            <td><input type="text" value="${row.courier   ||''}" placeholder="Courier"></td>
            <td><input type="url"  value="${row.slipLink || ''}" placeholder="https://… tracking / courier link"></td>
            <td>
              <div class="uploader">
                <input type="file" accept=".png,.jpg,.jpeg,.pdf">
                <button class="btn btn-small" type="button">Get Courier Details</button>
              </div>
              <div class="muted" style="margin-top:4px;font-size:11px">Suggestions will appear below</div>
              <div class="muted" style="margin-top:4px;font-size:11px" data-suggestions></div>
              <img class="preview" style="display:none">
            </td>
            <td>
              <div class="uploader">
                <input type="file" accept=".pdf,.png,.jpg,.jpeg">
              </div>
              <div class="muted" style="margin-top:4px;font-size:11px" data-eway-hint></div>
            </td>
            <td>
              <button class="btn btn-small" type="button">Update &amp; Email</button>
              <div class="muted" style="font-size:11px;margin-top:4px"></div>
            </td>
            <td>${row.emailSentAt || ''}</td>`;

          const tds      = tr.querySelectorAll('td');
          const inpTid   = tds[4].querySelector('input');
          const inpCour  = tds[5].querySelector('input');
          const inpUrl   = tds[6].querySelector('input');

          // Courier slip cell (column 8)
          const slipCell = tds[7];
          const fileSlip = slipCell.querySelector('input[type=file]');
          const btnOcr   = slipCell.querySelector('button');
          const prev     = slipCell.querySelector('img.preview');
          const sugg     = slipCell.querySelector('[data-suggestions]');

          // E-way bill cell (column 9)
          const ewayCell      = tds[8];
          const ewayFileInput = ewayCell.querySelector('input[type=file]');
          const ewayHint      = ewayCell.querySelector('[data-eway-hint]');

          // Action cell (column 10)
          const btnSave = tds[9].querySelector('button');
          const msg     = tds[9].querySelector('div');

          // ---- Grey out Update & Email if Email Sent At has value ----
          if (row.emailSentAt && String(row.emailSentAt).trim() !== '') {
            btnSave.disabled = true;
            btnSave.textContent = 'Email Sent';
          }

          // ---- Enable / disable E-way column based on E-Way flag ----
          const ewayFlag     = (row.eway || '').toString().trim().toUpperCase();
          const ewayRequired = (ewayFlag === 'YES');

          if (!ewayRequired) {
            ewayFileInput.disabled = true;
            ewayHint.textContent   = 'Not required (dispatch below ₹50,000).';
          } else {
            ewayFileInput.disabled = false;
            ewayHint.textContent   = 'Required (Total incl. GST above ₹50,000).';
          }

          // ----- OCR ON COURIER SLIP -----
          btnOcr.onclick = async ()=>{
            if(!fileSlip.files[0]){ alert('Choose courier slip image/PDF'); return; }
            const f = fileSlip.files[0];

            if(f.type.startsWith('image/')){
              const url=URL.createObjectURL(f);
              prev.src=url; prev.style.display='block';
            } else {
              prev.style.display='none';
            }

            btnOcr.disabled=true; btnOcr.textContent='OCR…';
            sugg.textContent = '';

            try{
              const dataUrl = await new Promise((res,rej)=>{
                const r=new FileReader();
                r.onload = ()=>res(r.result);
                r.onerror= rej;
                r.readAsDataURL(f);
              });
              google.script.run
                .withSuccessHandler(r=>{
                  btnOcr.disabled=false; btnOcr.textContent='Get Courier Details';
                  if(r && r.ok){
                    if(r.tracking) inpTid.value = r.tracking;
                    if(r.courier)  inpCour.value = r.courier;
                    if(Array.isArray(r.candidates) && r.candidates.length){
                      sugg.textContent = 'Suggestions: ' + r.candidates.join(', ');
                    }
                  } else {
                    alert('OCR error: '+(r && r.msg ? r.msg : 'unknown'));
                  }
                })
                .withFailureHandler(err=>{
                  btnOcr.disabled=false; btnOcr.textContent='Get Courier Details';
                  alert('Server error: '+err.message);
                })
                .api_ocrVisionOnly(f.name, dataUrl);
            }catch(e){
              btnOcr.disabled=false; btnOcr.textContent='Get Courier Details';
              alert('Read error: '+e.message);
            }
          };

          // ----- UPDATE & EMAIL -----
          btnSave.onclick = ()=>{
            if (btnSave.disabled) return; // safety

            const tid = inpTid.value.trim();
            if (!tid) { alert('Enter tracking ID'); return; }

            btnSave.disabled = true;
            msg.textContent  = 'Sending…';

            const courier   = inpCour.value.trim();
            const docketUrl = inpUrl.value.trim();

            const slipFile  = fileSlip.files[0]   || null;
            const ewayFile  = ewayFileInput.files[0] || null;

            function callServer(payload){
              google.script.run
                .withSuccessHandler(r=>{
                  if (r && r.ok){
                    msg.textContent = 'Updated & emailed.';
                    btnSave.textContent = 'Email Sent';
                    btnSave.disabled = true;
                  } else {
                    msg.textContent = 'Failed: ' + (r && r.msg ? r.msg : 'unknown');
                    btnSave.disabled = false;
                  }
                })
                .withFailureHandler(err=>{
                  btnSave.disabled = false;
                  msg.textContent = 'Server error: ' + err.message;
                })
                .api_updateTrackingAndNotify_v4(payload);
            }

            // Helper: read a file to dataUrl
            function readAsDataUrl(file, cb, onError){
              const reader = new FileReader();
              reader.onload  = e => cb(e.target.result);
              reader.onerror = e => onError(e);
              reader.readAsDataURL(file);
            }

            // Case 1: no files at all
            if (!slipFile && !ewayFile){
              const payload = {
                rowIndex:   row.rowIndex,
                trackingId: tid,
                courier:    courier,
                docketUrl:  docketUrl,
                slipFile:   null,
                ewayFile:   null
              };
              callServer(payload);
              return;
            }

            const basePayload = {
              rowIndex:   row.rowIndex,
              trackingId: tid,
              courier:    courier,
              docketUrl:  docketUrl,
              slipFile:   null,
              ewayFile:   null
            };

            // Case 2: only slip
            if (slipFile && !ewayFile){
              readAsDataUrl(slipFile, dataUrl=>{
                basePayload.slipFile = {
                  name: slipFile.name,
                  dataUrl: dataUrl
                };
                callServer(basePayload);
              }, e=>{
                btnSave.disabled = false;
                msg.textContent = 'Slip read error: ' + e.message;
              });
              return;
            }

            // Case 3: only e-way
            if (!slipFile && ewayFile){
              readAsDataUrl(ewayFile, dataUrl=>{
                basePayload.ewayFile = {
                  name: ewayFile.name,
                  dataUrl: dataUrl
                };
                callServer(basePayload);
              }, e=>{
                btnSave.disabled = false;
                msg.textContent = 'E-way bill read error: ' + e.message;
              });
              return;
            }

            // Case 4: both slip and e-way present
            readAsDataUrl(slipFile, slipData=>{
              basePayload.slipFile = {
                name: slipFile.name,
                dataUrl: slipData
              };
              readAsDataUrl(ewayFile, ewayData=>{
                basePayload.ewayFile = {
                  name: ewayFile.name,
                  dataUrl: ewayData
                };
                callServer(basePayload);
              }, e=>{
                btnSave.disabled = false;
                msg.textContent = 'E-way bill read error: ' + e.message;
              });
            }, e=>{
              btnSave.disabled = false;
              msg.textContent = 'Slip read error: ' + e.message;
            });
          };

          $('tbody').appendChild(tr);
        });
      })
      .withFailureHandler(err=>{
        console.log('api_listDispatchesByRange_v1 failure:', err);
        $('status').textContent='Error: '+err.message;
        $('status').className   ='muted';
        alert('Error loading dispatches: '+err.message);
      })
      .api_listDispatchesByRange_v1(from, to);
  }

  // ============ CHAT ASSISTANT LOGIC ============
  const CHAT_CONFIG = {
    scope: 'tracking',
    page:  'tracking'
  };

  function appendMessage(role, text){
    const body = $('chatBody');
    const wrap = document.createElement('div');
    wrap.className = 'chat-message ' + (role === 'user' ? 'user' : 'assistant');

    const bubble = document.createElement('div');
    bubble.className = 'chat-bubble';
    bubble.innerHTML = text.replace(/\n/g,'<br>');
    wrap.appendChild(bubble);
    body.appendChild(wrap);
    body.scrollTop = body.scrollHeight;
  }

  function openChat(){
    $('chatPanel').classList.remove('hidden');
    $('chatInput').focus();
  }

  function closeChat(){
    $('chatPanel').classList.add('hidden');
  }

  function sendChat(){
    const input = $('chatInput');
    const status = $('chatStatus');
    const msg = input.value.trim();
    if (!msg) return;

    if (!hasGoogle()){
      alert('Server not ready yet. Please try again in a moment.');
      return;
    }

    appendMessage('user', msg);
    input.value = '';
    status.textContent = 'Thinking about your question…';
    $('chatSend').disabled = true;

    const payload = {
      message:   msg,
      scope:     CHAT_CONFIG.scope,
      page:      CHAT_CONFIG.page,
      url:       (typeof location !== 'undefined' ? location.href : ''),
      timestamp: new Date().toISOString()
    };

    google.script.run
      .withSuccessHandler(res=>{
        $('chatSend').disabled = false;
        if (!res){
          status.textContent = 'No response from assistant.';
          return;
        }
        if (res.ok === false){
          status.textContent = 'Error: ' + (res.msg || 'Unknown');
          return;
        }
        status.textContent = 'Online';
        const ans = res.answer || 'No answer received.';
        appendMessage('assistant', ans);
      })
      .withFailureHandler(err=>{
        $('chatSend').disabled = false;
        status.textContent = 'Server error: ' + (err && err.message ? err.message : 'Unknown');
      })
      .api_dispatchChatAssistant(payload);
  }

  function init(){
    const todayIST = new Date().toLocaleDateString('en-CA',{timeZone:'Asia/Kolkata'});
    $('fromDate').value = todayIST;
    $('toDate').value   = todayIST;

    $('loadBtn').onclick = function(){
      if(!hasGoogle()){
        alert('Page is still initialising, please try again in a moment.');
        return;
      }
      loadRange();
    };

    // Chat events
    $('chatLauncher').onclick = openChat;
    $('chatClose').onclick    = closeChat;
    $('chatSend').onclick     = sendChat;
    $('chatInput').addEventListener('keydown', function(e){
      if (e.key === 'Enter' && !e.shiftKey){
        e.preventDefault();
        sendChat();
      }
    });
  }

  window.onload = init;
</script>
