  <script>
    /* -----------------------------------------
       GLOBAL STATE
       --------------------------------------*/
    var CLIENTS = [];
    var INSTALLERS = [];
    var PROJECTS = []; // Global store for current client's projects
    var PRODUCTS = [];
    var invoiceFile = null;

    /* -----------------------------------------
       INIT
       --------------------------------------*/
    function init() {
      // Set today's date
      var d = new Date();
      var yyyy = d.getFullYear();
      var mm = ('0' + (d.getMonth() + 1)).slice(-2);
      var dd = ('0' + d.getDate()).slice(-2);
      document.getElementById('field-date').value = yyyy + '-' + mm + '-' + dd;

      // Reset ship-to mode to "client"
      var radios = document.querySelectorAll('input[name="shipto"]');
      if (radios && radios.length) {
        radios[0].checked = true; // client
      }
      syncShipToModeUI();

      // File input
      var f = document.getElementById('field-invoice-file');
      f.addEventListener('change', function (e) {
        invoiceFile = e.target.files && e.target.files[0] ? e.target.files[0] : null;
        document.getElementById('ocr-status').textContent = invoiceFile
          ? 'File selected: ' + invoiceFile.name
          : '';
      });

      // Reset items table
      var tbody = document.getElementById('items-tbody');
      tbody.innerHTML = '';
      addItemRow();

      // Load clients + installers from server
      loadClients();
      loadInstallers();
    }

    document.addEventListener('DOMContentLoaded', init);

    /* -----------------------------------------
       SHIP-TO MODE HANDLING
       --------------------------------------*/
    function getShipToMode() {
      var radios = document.querySelectorAll('input[name="shipto"]');
      for (var i = 0; i < radios.length; i++) {
        if (radios[i].checked) return radios[i].value;
      }
      return 'client';
    }

    function syncShipToModeUI() {
      var mode = getShipToMode();
      var installerBlock = document.getElementById('installer-block');
      if (!installerBlock) return;

      if (mode === 'installer') {
        installerBlock.classList.remove('hidden');
      } else {
        installerBlock.classList.add('hidden');
        // When shipping to client, re-apply client details as ship-to (if available)
        var cname = document.getElementById('field-client-name').value || '';
        if (cname) {
          var c = findClientByName(cname);
          if (c) {
            fillShipToFromClient(c);
          }
        }
      }
    }

    function onShipToModeChange() {
      syncShipToModeUI();
    }

    /* -----------------------------------------
       LOAD CLIENTS
       --------------------------------------*/
    function loadClients() {
      if (typeof google === 'undefined' || !google.script || !google.script.run) return;
      google.script.run
        .withSuccessHandler(function (list) {
          CLIENTS = list || [];
          // Auto-refresh dropdown if user is currently focused on the field
          var input = document.getElementById('field-client-name');
          if (input && document.activeElement === input) {
            renderClientSuggestions(input.value.trim());
          }
        })
        .withFailureHandler(function (err) {
          console.error('api_getClients error: ', err);
        })
        .api_getClients();
    }

    /* -----------------------------------------
       LOAD INSTALLERS
       --------------------------------------*/
    function loadInstallers() {
      if (typeof google === 'undefined' || !google.script || !google.script.run) return;
      google.script.run
        .withSuccessHandler(function (res) {
          if (!res || !res.ok) return;
          INSTALLERS = res.installers || [];
          populateInstallerSelect();
        })
        .withFailureHandler(function (err) {
          console.error('api_getInstallers error: ', err);
        })
        .api_getInstallers();
    }

    function populateInstallerSelect() {
      var sel = document.getElementById('field-installer');
      if (!sel) return;

      // Reset
      sel.innerHTML = '';
      var optDefault = document.createElement('option');
      optDefault.value = '';
      optDefault.textContent = 'Select installer…';
      sel.appendChild(optDefault);

      INSTALLERS.forEach(function (inst) {
        var opt = document.createElement('option');
        opt.value = inst.name;
        opt.textContent = inst.name + (inst.city ? ' — ' + inst.city : '');
        sel.appendChild(opt);
      });
    }

    function onInstallerChange() {
      var name = document.getElementById('field-installer').value || '';
      if (!name) {
        document.getElementById('field-poc').value = '';
        document.getElementById('field-phone').value = '';
        document.getElementById('field-email').value = '';
        document.getElementById('field-address').value = '';
        return;
      }

      var inst = null;
      for (var i = 0; i < INSTALLERS.length; i++) {
        if ((INSTALLERS[i].name || '').toLowerCase() === name.toLowerCase()) {

        })

    }

    function parseCityState(addr) {
      if (!addr) return '';
      var s = String(addr).trim();
      var sLower = s.toLowerCase();

      // 1. Clean Pincodes / India
      // Remove generic 6-digit or 5-7 digit codes at the end
      s = s.replace(/(?:(?:Pin|Zip|Code|Pincode)[\s.:-]*)?[-–\s]*\d{5,7}\s*$/i, '');
      s = s.replace(/,?\s*(?:India|IN)\s*$/i, '');
      var cleaned = s.replace(/[,.-]+$/, '').trim();

      // BRUTE FORCE SEARCH
      // Find the KNOWN city that appears LAST in the string
      var matchedCity = null;
      var matchedCityIndex = -1;

      for (var city in INDIA_GEO.cities) {
        var idx = sLower.indexOf(city.toLowerCase());
        // We want the last occurrence if multiple, but usually just one unique city.
        // Actually we want the right-most city in the string to be safe.
        var lastIdx = sLower.lastIndexOf(city.toLowerCase());

        if (lastIdx !== -1) {
          // Heuristic: Prefer longer match if index is same? 
          // Or just take the one that is furthest to the right (most specific location part usually at end).
          // But beware of "New Delhi" vs "Delhi". 
          // If we matched "Delhi" at index 10 and "New Delhi" at index 6, "New Delhi" spans up to 15.

          // Let's store candidates and pick best.
          // For simple brute force: keep the one that spans furthest right.
          var endPos = lastIdx + city.length;
          // Update if this city ends later than current best
          if (!matchedCity || endPos > (matchedCityIndex + matchedCity.length)) {
            matchedCity = city;
            matchedCityIndex = lastIdx;
          }
          // If ends at same place, pick longer name (e.g. New Delhi over Delhi)
          else if (endPos === (matchedCityIndex + matchedCity.length)) {
            if (city.length > matchedCity.length) {
              matchedCity = city;
              matchedCityIndex = lastIdx;
            }
          }
        }
      }

      // Find the KNOWN state that appears LAST in the string
      // (ignoring the one implied by city for a moment, let's see what's written)
      var matchedState = null;
      var matchedStateIndex = -1;

      for (var k = 0; k < INDIA_GEO.states.length; k++) {
        var st = INDIA_GEO.states[k];
        var idx = sLower.lastIndexOf(st.toLowerCase());
        if (idx !== -1) {
          var endPos = idx + st.length;
          if (!matchedState || endPos > (matchedStateIndex + matchedState.length)) {
            matchedState = st;
            matchedStateIndex = idx;
          }
        }
      }

      // DECISION LOGIC

      // 1. KNOWN CITY FOUND
      if (matchedCity) {
        var impliedState = INDIA_GEO.cities[matchedCity];
        if (matchedState) {
          return matchedCity + ', ' + matchedState;
        } else {
          return matchedCity + ', ' + impliedState;
        }
      }

      // 2. ONLY KNOWN STATE FOUND (City unknown)
      if (matchedState) {
        // User Requirement: "if found city or state... no need to fetch full address"
        // Heuristic: The word/token immediately BEFORE the state is likely the city.

        var beforeState = s.substring(0, matchedStateIndex).trim();
        // Remove trailing separators
        beforeState = beforeState.replace(/[,.\s-]+$/, '');

        // Get the last "chunk" (usually separated by comma)
        var chunks = beforeState.split(',');
        var lastChunk = chunks[chunks.length - 1].trim();

        // Sanity check: reasonable length and contains letters
        if (lastChunk.length > 0 && lastChunk.length < 30) {
          // Strip trailing dash if any
          lastChunk = lastChunk.replace(/[-]+$/, '').trim();

          if (/[a-zA-Z]/.test(lastChunk)) {
            return lastChunk + ', ' + matchedState;
          }
        }

        // Fallback: If heuristic failed, return JUST the state.
        return matchedState;
      }

      // 3. NOTHING FOUND -> Return Full Address (Cleaned)
      // Legacy comma heuristic as last resort before giving up?
      var parts = cleaned.split(',').map(function (p) { return p.trim() }).filter(function (p) { return p });
      if (parts.length >= 2) {
        var last = parts[parts.length - 1];
        var dashes = last.split(/\s+[-–]\s+/);
        if (dashes.length >= 2) {
          return dashes[dashes.length - 2] + ', ' + dashes[dashes.length - 1];
        }
        // Heuristic if simple comma separation exists
        return parts[parts.length - 2] + ', ' + parts[parts.length - 1];
      }

      return cleaned;
    }

    function renderClientSuggestions(query) {
      var box = document.getElementById('client-suggestions');
      box.innerHTML = '';

      var showAll = !query; // Show all if input is empty
      var qLower = (query || '').toLowerCase();

      var matches = [];
      if (showAll) {
        matches = CLIENTS;
      } else {
        matches = CLIENTS.filter(function (c) {
          return String(c.name || '').toLowerCase().indexOf(qLower) !== -1;
        });
      }

      // HANDLE EMPTY STATE (Loading or No Data)
      if (showAll && !matches.length) {
        var msgDiv = document.createElement('div');
        msgDiv.className = 'suggestion-item';
        msgDiv.style.color = '#9ca3af';
        msgDiv.style.cursor = 'default';
        if (CLIENTS.length === 0) {
          msgDiv.textContent = 'Loading clients...';
        } else {
          msgDiv.textContent = 'No clients found.';
        }
        box.appendChild(msgDiv);
        box.classList.remove('hidden');
        return;
      }

      // Exact match check (only relevant if filtered)
      var exactExists = false;
      if (!showAll) {
        exactExists = CLIENTS.some(function (c) {
          return String(c.name || '').trim().toLowerCase() === qLower;
        });
      }

      // If no matches and not showing all, maybe show "Add"
      if (!matches.length && !showAll) {
        if (query.length > 1 && !exactExists) {
          renderAddClientOption(box, query);
          box.classList.remove('hidden');
        } else {
          box.classList.add('hidden');
        }
        return;
      }

      // Render matches
      matches.forEach(function (c) {
        var item = document.createElement('div');
        item.className = 'suggestion-item client-suggestion-item';

        var nameEl = document.createElement('div');
        nameEl.className = 'suggestion-name client-suggestion-name';
        nameEl.textContent = c.name || '';

        var metaEl = document.createElement('div');
        metaEl.className = 'client-suggestion-meta';

        // FORMAT: Client Name + POC Name + Address (City, State)
        // No phone number.
        var cityState = parseCityState(c.address || '');
        var parts = [];
        if (c.poc) parts.push(c.poc);
        if (cityState) parts.push(cityState);

        metaEl.textContent = parts.join(' • ');

        item.appendChild(nameEl);
        item.appendChild(metaEl);
        item.onclick = function () {
          selectExistingClient(c);
        };
        box.appendChild(item);
      });

      // Append "Add new" option at the bottom if applicable
      if (!showAll && query.length > 1 && !exactExists) {
        renderAddClientOption(box, query);
      }

      box.classList.remove('hidden');
    }

    function addItemRow(desc, qty, amount) {
      var tbody = document.getElementById('items-tbody');
      var rowIndex = tbody.rows.length;
      var tr = document.createElement('tr');

      var tdIdx = document.createElement('td');
      tdIdx.textContent = rowIndex + 1;

      var tdDesc = document.createElement('td');
      var inpDesc = document.createElement('input');
      inpDesc.type = 'text';
      inpDesc.setAttribute('list', 'products-datalist'); // Enable autocomplete
      inpDesc.value = desc || '';
      tdDesc.appendChild(inpDesc);

      var tdQty = document.createElement('td');
      var inpQty = document.createElement('input');
      inpQty.type = 'number';
      inpQty.min = '0';
      inpQty.step = '1';
      inpQty.value = qty != null ? qty : '';
      inpQty.oninput = recalcTotals;
      tdQty.appendChild(inpQty);

      var tdAmt = document.createElement('td');
      var inpAmt = document.createElement('input');
      inpAmt.type = 'number';
      inpAmt.min = '0';
      inpAmt.step = '0.01';
      inpAmt.value = amount != null ? amount : '';
      inpAmt.oninput = recalcTotals;
      tdAmt.appendChild(inpAmt);

      var tdRemove = document.createElement('td');
      tdRemove.className = 'right';
      var btnR = document.createElement('button');
      btnR.type = 'button';
      btnR.className = 'btn btn-ghost btn-small';
      btnR.textContent = '×';
      btnR.onclick = function () {
        tbody.removeChild(tr);
        renumberItems();
        recalcTotals();
      };
      tdRemove.appendChild(btnR);

      tr.appendChild(tdIdx);
      tr.appendChild(tdDesc);
      tr.appendChild(tdQty);
      tr.appendChild(tdAmt);
      tr.appendChild(tdRemove);

      tbody.appendChild(tr);
      recalcTotals();
    }

    function renumberItems() {
      var tbody = document.getElementById('items-tbody');
      for (var i = 0; i < tbody.rows.length; i++) {
        tbody.rows[i].cells[0].textContent = i + 1;
      }
    }

    function getItemsData() {
      var tbody = document.getElementById('items-tbody');
      var items = [];
      for (var i = 0; i < tbody.rows.length; i++) {
        var row = tbody.rows[i];
        var desc = row.cells[1].querySelector('input').value || '';
        var qtyVal = row.cells[2].querySelector('input').value || '';
        var amtVal = row.cells[3].querySelector('input').value || '';
        if (!desc && !qtyVal && !amtVal) continue;
        var qty = Number(qtyVal || 0);
        var amt = Number(amtVal || 0);
        items.push({ desc: desc, qty: qty, amount: amt });
      }
      return items;
    }

    function recalcTotals() {
      var items = getItemsData();
      var totalQty = 0;
      var subtotal = 0;
      items.forEach(function (it) {
        totalQty += Number(it.qty || 0);
        subtotal += Number(it.amount || 0);
      });
      var gst = subtotal * 0.18;
      var grand = subtotal + gst;

      document.getElementById('chip-total-qty').textContent =
        'Total Qty: ' + totalQty;
      document.getElementById('chip-subtotal').textContent =
        'Subtotal: ₹' + subtotal.toFixed(2);
      document.getElementById('chip-gst').textContent =
        'GST 18%: ₹' + gst.toFixed(2);
      document.getElementById('chip-grand').textContent =
        'Total (incl. GST): ₹' + grand.toFixed(2);

      // 🔴 NEW: Toggle E-way bill warning if Total incl. GST ≥ 50,000
      var ew = document.getElementById('eway-warning');
      if (ew) {
        if (grand >= 50000) {
          ew.style.display = 'block';
        } else {
          ew.style.display = 'none';
        }
      }
    }

    function runOcr() {
      var statusEl = document.getElementById('ocr-status');
      statusEl.style.color = '#6b7280';
      if (!invoiceFile) {
        statusEl.textContent = 'Please upload an invoice file first.';
        return;
      }
      if (!(typeof FileReader !== 'undefined')) {
        statusEl.textContent = 'FileReader not supported in this environment.';
        return;
      }

      statusEl.textContent = 'Running OCR (Doc AI)…';
      var reader = new FileReader();
      reader.onload = function (e) {
        var dataUrl = e.target.result;
        var filename = invoiceFile.name || 'invoice';

        if (typeof google === 'undefined' || !google.script || !google.script.run) {
          statusEl.textContent = 'google.script.run not available in preview.';
          return;
        }

        google.script.run
          .withSuccessHandler(function (res) {
            handleOcrResult(res, filename, dataUrl, true);
          })
          .withFailureHandler(function () {
            statusEl.textContent =
              'Doc AI failed, falling back to Vision OCR…';
            google.script.run
              .withSuccessHandler(function (res2) {
                handleOcrResult(res2, filename, dataUrl, false);
              })
              .withFailureHandler(function (err2) {
                statusEl.style.color = '#b91c1c';
                statusEl.textContent =
                  'OCR failed: ' + (err2 && err2.message ? err2.message : err2);
              })
              .api_parseInvoice(filename, dataUrl);
          })
          .api_parseInvoice_docai(filename, dataUrl);
      };
      reader.readAsDataURL(invoiceFile);
    }

    function handleOcrResult(res, filename, dataUrl, wasDocAI) {
      var statusEl = document.getElementById('ocr-status');
      if (!res || res.ok === false) {
        statusEl.style.color = '#b91c1c';
        statusEl.textContent =
          (wasDocAI ? 'Doc AI error: ' : 'OCR error: ') +
          (res && res.msg ? res.msg : 'Unknown error');
        return;
      }

      var items = res.items || [];
      if (!items.length) {
        statusEl.style.color = '#6b7280';
        statusEl.textContent =
          (wasDocAI ? 'Doc AI ' : 'OCR ') +
          'did not find structured line items. You can fill manually.';
        return;
      }

      var tbody = document.getElementById('items-tbody');
      tbody.innerHTML = '';
      items.forEach(function (it) {
        addItemRow(it.desc || '', it.qty || 0, it.amount || 0);
      });
      recalcTotals();
      statusEl.style.color = '#15803d';
      statusEl.textContent =
        'Items filled from ' + (wasDocAI ? 'Doc AI' : 'Vision') + ' OCR.';
    }

    function submitDispatch() {
      hideStatus();
      var btn = document.getElementById('btn-submit');
      btn.disabled = true;

      var dateStr = document.getElementById('field-date').value || '';
      var clientName = document.getElementById('field-client-name').value || '';
      var projectName = document.getElementById('field-project').value || '';
      var installerName = document.getElementById('field-installer').value || '';
      var poc = document.getElementById('field-poc').value || '';
      var phone = document.getElementById('field-phone').value || '';
      var email = document.getElementById('field-email').value || '';
      var address = document.getElementById('field-address').value || '';
      var items = getItemsData();

      if (!dateStr || !clientName) {
        showError('Please fill Date and Select Client before submitting.');
        btn.disabled = false;
        return;
      }
      if (!items.length) {
        showError('Please add at least one line item.');
        btn.disabled = false;
        return;
      }

      var payload = {
        date: dateStr,
        clientName: clientName,
        clientAddress: address,
        poc: poc,
        phone: phone,
        clientEmail: email,
        projectDetails: projectName,
        items: items,
        installerName: installerName
      };

      if (typeof google === 'undefined' || !google.script || !google.script.run) {
        showError('google.script.run not available in preview.');
        btn.disabled = false;
        return;
      }

      google.script.run
        .withSuccessHandler(function (res) {
          btn.disabled = false;
          if (!res || res.ok === false) {
            showError('Server error: ' + (res && res.msg ? res.msg : 'Unknown error'));
            return;
          }
          var dispatchId = res.dispatchId || '';
          var url = res.combinedPdfUrl || '';
          showSuccess(dispatchId, url);
        })
        .withFailureHandler(function (err) {
          btn.disabled = false;
          showError(
            'Submit failed: ' + (err && err.message ? err.message : err)
          );
        })
        .api_submitDispatch(payload);
    }

    function showSuccess(dispatchId, url) {
      var box = document.getElementById('submit-status');
      var title = document.getElementById('submit-status-title');
      var text = document.getElementById('submit-status-text');
      var link = document.getElementById('btn-download-pdf');

      title.textContent = 'Dispatch generated successfully.';
      text.textContent =
        'Dispatch ID: ' +
        dispatchId +
        (url ? ' • Combined PDF is ready to download.' : '');

      if (url) {
        link.href = url;
        link.classList.remove('hidden');
        // Auto-open PDF in new tab on successful submission
        try {
          window.open(url, '_blank');
        } catch (e) {
          console.warn('Popup blocked while opening PDF:', e);
        }
      } else {
        link.href = '#';
        link.classList.add('hidden');
      }

      box.style.display = 'flex';
      box.classList.remove('hidden');
      document.getElementById('error-status').classList.add('hidden');
    }

    function showError(msg) {
      var box = document.getElementById('error-status');
      var text = document.getElementById('error-status-text');
      text.textContent = msg || 'Something went wrong.';
      box.style.display = 'flex';
      box.classList.remove('hidden');
      document.getElementById('submit-status').classList.add('hidden');
    }

    function hideStatus() {
      document.getElementById('submit-status').classList.add('hidden');
      document.getElementById('error-status').classList.add('hidden');
    }
