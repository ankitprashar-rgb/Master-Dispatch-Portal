<script>
  function $(id){ return document.getElementById(id); }

  function hasGoogle(){
    return (typeof google !== 'undefined') &&
           google.script &&
           google.script.run;
  }

  // ---------- SUMMARY CHIP ----------
  function updateSummaryChip(summary, rows){
    var chip = $('summaryChip');
    if (!chip) return;

    var list = Array.isArray(rows) ? rows : [];

    var count = 0;
    var totalAmount = 0;

    if (summary && typeof summary === 'object') {
      if (typeof summary.count === 'number') {
        count = summary.count;
      }
      if (typeof summary.totalAmount === 'number') {
        totalAmount = summary.totalAmount;
      }
    }

    // Fallback to rows if summary not provided by backend
    if (!count) {
      count = list.length;
    }
    if (!totalAmount) {
      for (var i = 0; i < list.length; i++) {
        var v = Number(list[i].amount || 0);
        if (!isNaN(v)) totalAmount += v;
      }
    }

    if (!count) {
      chip.classList.add('hidden');
      chip.textContent = '';
      return;
    }

    var formattedTotal = '₹' + Number(totalAmount || 0).toLocaleString('en-IN', {
      maximumFractionDigits: 0
    });

    chip.textContent = count + ' dispatch' + (count > 1 ? 'es' : '') + ' • ' + formattedTotal;
    chip.classList.remove('hidden');
  }

  // ---------- CLIENTS ----------
  function loadClients() {
    if (!hasGoogle()) return;
    google.script.run
      .withSuccessHandler(function(list){
        var s = $('client');
        s.innerHTML = '<option value="">(All)</option>';
        if (list && list.length) {
          for (var i=0; i<list.length; i++){
            var c = list[i];
            var o = document.createElement('option');
            o.value = c.name;
            o.textContent = c.name;
            s.appendChild(o);
          }
        }
      })
      .withFailureHandler(function(err){
        console.log('Clients error:', err);
      })
      .api_getClients();
  }

  // ---------- RENDER TABLE ----------
  function renderRows(rows) {
    var tb = $('tbl').getElementsByTagName('tbody')[0];
    tb.innerHTML = '';
    if (!rows || !rows.length) {
      tb.innerHTML = '<tr><td colspan="15">No records found.</td></tr>';
      return;
    }
    for (var i=0; i<rows.length; i++){
      var r = rows[i];
      var tr = document.createElement('tr');

      var addr = (r.address || '');
      addr = addr.toString().replace(/\n/g,'<br>');

      var lbl   = r.label   ? '<a href="'+r.label+'" target="_blank">Open</a>'   : '';
      var chal  = r.challan ? '<a href="'+r.challan+'" target="_blank">Open</a>' : '';
      var slip  = r.slip    ? '<a href="'+r.slip+'" target="_blank">Open</a>'    : '';
      var amt   = (r.amount !== '' && r.amount != null) ? ('₹' + r.amount) : '';

      tr.innerHTML =
        '<td>' + (r.date    || '') + '</td>' +
        '<td>' + (r.client  || '') + '</td>' +
        '<td>' + addr              + '</td>' +
        '<td>' + (r.poc     || '') + '</td>' +
        '<td>' + (r.project || '') + '</td>' +
        '<td>' + (r.qty     || '') + '</td>' +
        '<td>' + amt               + '</td>' +
        '<td>' + (r.eway    || '') + '</td>' +
        '<td>' + lbl               + '</td>' +
        '<td>' + chal              + '</td>' +
        '<td>' + (r.did     || '') + '</td>' +
        '<td>' + (r.track   || '') + '</td>' +
        '<td>' + (r.courier || '') + '</td>' +
        '<td>' + slip              + '</td>' +
        '<td>' + (r.mailed  || '') + '</td>';

      tb.appendChild(tr);
    }
  }

  // ---------- SEARCH ----------
  function runSearch() {
    if (!hasGoogle()) {
      alert('Page is still initialising, please wait a moment and try again.');
      return;
    }

    var from = $('dateFrom').value;
    var to   = $('dateTo').value;
    var c    = $('client').value ? $('client').value.trim() : '';

    $('status').textContent = 'Searching…';
    $('status').className   = 'muted dots';

    google.script.run
      .withSuccessHandler(function(res){
        $('status').textContent = '';
        $('status').className   = 'muted';

        var rows = [];
        var summary = null;

        // Handle different backend shapes gracefully
        if (res && res.ok === false) {
          alert('Server error while fetching summary: ' + (res.msg || 'Unknown'));
          updateSummaryChip(null, []);
          renderRows([]);
          return;
        }

        if (Array.isArray(res)) {
          rows = res;
        } else if (res && Array.isArray(res.rows)) {
          rows = res.rows;
          if (res.summary) summary = res.summary;
        }

        updateSummaryChip(summary, rows);
        renderRows(rows);
      })
      .withFailureHandler(function(err){
        console.log('Summary error:', err);
        $('status').textContent = 'Error: ' + err.message;
        $('status').className   = 'muted';
        updateSummaryChip(null, []);
      })
      .api_getSummary(from, to, c);   // backend: (fromDate, toDate, client)
  }

  // ---------- INIT DOM ----------
  function initDom() {
    try {
      // Default: current month range (1st → today in IST)
      var now = new Date();
      var year  = now.getFullYear();
      var month = ('0' + (now.getMonth() + 1)).slice(-2);
      var todayIST = new Date().toLocaleDateString('en-CA', { timeZone:'Asia/Kolkata' });
      var firstOfMonth = year + '-' + month + '-01';

      $('dateFrom').value = firstOfMonth;
      $('dateTo').value   = todayIST;
    } catch(e) {
      console.log('Date init error:', e);
    }
    $('btnSearch').onclick = runSearch;
  }

  // ---------- WAIT FOR google.script.run ----------
  function bootWithGoogle(tries) {
    if (typeof tries === 'undefined') tries = 0;
    if (hasGoogle()) {
      loadClients();
      runSearch(); // auto-load range
      return;
    }
    if (tries > 40) {
      console.log('google.script.run never became available');
      $('status').textContent = 'Could not connect to server (google.script.run not ready).';
      return;
    }
    setTimeout(function(){ bootWithGoogle(tries+1); }, 250);
  }

  // ---------- GLOBAL INIT ----------
  window.onload = function(){
    initDom();
    bootWithGoogle(0);
  };
</script>

(function(){
  const btn  = document.getElementById('ide-chat-btn');
  const box  = document.getElementById('ide-chat-box');
  const send = document.getElementById('ide-chat-send');
  const out  = document.getElementById('ide-chat-answer');

  btn.onclick = ()=> box.style.display = (box.style.display==='none' || box.style.display==='') ? 'block' : 'none';

  send.onclick = function(){
    const msg = document.getElementById('ide-chat-msg').value.trim();
    if(!msg){ alert("Enter a question"); return; }

    out.innerHTML = "Thinking…";

    // 🔥 FIXED SCOPE PER PAGE — CHANGE PER FILE
    const payload = {
      message: msg,
      scope:   CHAT_SCOPE,    // defined below
      page:    CHAT_SCOPE,
      url:     window.location.href
    };

    google.script.run
      .withSuccessHandler(res=>{
        if(res && res.ok){
          out.innerHTML = res.answer;
        } else {
          out.innerHTML = "Error: " + (res.msg || 'Unknown');
        }
      })
      .withFailureHandler(err=>{
        out.innerHTML = "Server Error: " + err.message;
      })
      .api_dispatchChatAssistant(payload);
  };
})();
</script>
